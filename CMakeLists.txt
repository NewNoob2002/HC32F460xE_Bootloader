cmake_minimum_required(VERSION 3.20)
option(BOOT_BUILD_HOST_TESTS "Build hardware-independent host tests" OFF)
option(BOOT_ENABLE_LOGGING "Enable SEGGER RTT and EasyLogger in Boot" ON)
option(APP_ENABLE_LOGGING "Enable SEGGER RTT and EasyLogger in App" ON)

if(BOOT_BUILD_HOST_TESTS)
    project(HC32F460_Firmware LANGUAGES C CXX)
else()
    project(HC32F460_Firmware LANGUAGES C CXX ASM)
endif()
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)

include(cmake/ProjectOptions.cmake)
add_subdirectory(Common)
add_subdirectory(App)

if(BOOT_BUILD_HOST_TESTS)
    enable_testing()
    add_subdirectory(Tests)
    return()
endif()

add_subdirectory(Platform)
add_subdirectory(Libraries)
add_subdirectory(Boot)
